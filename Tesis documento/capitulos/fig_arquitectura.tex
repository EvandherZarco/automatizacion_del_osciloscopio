% Figura: arquitectura de módulos del sistema de automatización
% Requiere en el preámbulo:
%   \usepackage{tikz}

% =========================================================

\begin{figure}[htbp]
  \centering
  \begin{tikzpicture}[
      font=\footnotesize,
      modulo/.style={
        draw, rounded corners=2pt, align=center,
        text width=34mm, minimum height=11mm, inner sep=2pt},
      hw/.style={modulo, fill=black!6},
      capa/.style={
        draw=black!35, dashed, rounded corners=4pt, inner sep=3mm, inner ysep=5mm},
      etiqueta/.style={font=\scriptsize\itshape, text=black!55},
      flecha/.style={-{Stealth[length=2mm]}, thick},
      linea/.style={thick},
      pasiva/.style={-{Stealth[length=2mm]}, densely dashed, black!55},
      nodo/.style={circle, fill, inner sep=0pt, minimum size=1.6mm},
      node distance=16mm and 10mm
    ]

    % --- Presentación ---
    \node[modulo] (gui) {Interfaz gráfica\\\scriptsize eventos y presentación};

    % --- Coordinación ---
    \node[modulo, below=of gui] (med)
      {Medición\\\scriptsize orquestación de la secuencia};
    \node[modulo, left=of med] (trig)
      {Disparo\\\scriptsize por tiempo o temperatura};
    \node[modulo, right=of med] (conn)
      {Conexión y monitoreo\\\scriptsize estado y reintentos};

    % --- Instrumentos ---
    \node[hw, below=of med] (oscil)
      {Osciloscopio\\\scriptsize VXI-11 sobre Ethernet};
    \node[hw, left=of oscil] (temp)
      {Temperatura\\\scriptsize ESP32 y sensores};
    \node[hw, right=of oscil] (laser)
      {Láser\\\scriptsize RS-232};

    % --- Datos y seguridad ---
    \node[modulo, below=of oscil] (store)
      {Almacenamiento\\\scriptsize registro de sesión};
    \node[modulo, left=of store] (viz)
      {Visualización\\\scriptsize consulta y exportación};
    \node[modulo, right=of store] (safe)
      {Modo seguro\\\scriptsize solo actúa sobre el láser};

    % --- Cajas de capa ---
    \begin{scope}[on background layer]
      \node[capa, fit=(trig)(med)(conn)] (cap_coord) {};
      \node[capa, fit=(temp)(oscil)(laser)] (cap_hw) {};
      \node[capa, fit=(viz)(store)(safe)] (cap_datos) {};
    \end{scope}

    \node[etiqueta, anchor=north east] at ([xshift=-1mm,yshift=-0.5mm]cap_coord.north east) {coordinación};
    \node[etiqueta, anchor=north east] at ([xshift=-1mm,yshift=-0.5mm]cap_hw.north east) {instrumentos};
    \node[etiqueta, anchor=north east] at ([xshift=-1mm,yshift=-0.5mm]cap_datos.north east) {datos};

    % --- Flujo de control desde la interfaz ---
    \draw[flecha] (gui) -- node[right, font=\scriptsize] {modo automático} (med);
    \draw[flecha] (gui.east) -| (conn.north);

    % --- Coordinación ---
    \draw[flecha] (med) -- (trig);
    \draw[flecha] ([xshift=-8mm]trig.south) -- ([xshift=-8mm]temp.north);

    % --- Bus de control de Medición hacia los instrumentos ---
    \coordinate (bus) at ([yshift=-8mm]med.south);
    \coordinate (carril) at ([xshift=5mm]oscil.east);
    \draw[linea] (med.south) -- (bus);
    \draw[linea] ([xshift=8mm]temp.north |- bus) -- (laser.north |- bus);
    \draw[flecha] ([xshift=8mm]temp.north |- bus) -- ([xshift=8mm]temp.north);
    \draw[flecha] (bus) -- (oscil.north);
    \draw[flecha] (laser.north |- bus) -- (laser.north);
    \draw[flecha] (carril |- bus) -- (carril |- safe.west) -- (safe.west);
    \node[nodo] at (bus) {};
    \node[nodo] at (carril |- bus) {};

    % --- Datos ---
    \draw[flecha] (oscil) -- (store);
    \draw[flecha] (store) -- (viz);
    \draw[flecha] (safe) -- (laser);

    % --- Vigilancia pasiva de los enlaces ---
    \draw[pasiva] (conn.east) -- ++(6mm,0) |- (cap_hw.east)
      node[pos=0.25, right, font=\scriptsize, text=black!55, align=left]
      {vigila los\\tres enlaces};

  \end{tikzpicture}
  \caption[Arquitectura de módulos del sistema de automatización]{%
    Arquitectura de módulos del sistema de automatización. Las flechas
    continuas indican control o transferencia de datos; la discontinua,
    la vigilancia pasiva del estado de conexión de los tres
    instrumentos. El módulo de medición coordina la secuencia y es el
    único que escribe en el registro de sesión. El modo seguro actúa
    exclusivamente sobre el láser: la configuración del osciloscopio no
    se modifica desde el programa.}
  \label{fig:arquitectura}
\end{figure}
